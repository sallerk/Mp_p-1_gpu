# ===========================================================================
#  Mp_p-1_gpu configuration
#
#  One "key = value" per line; '#' starts a comment. Numbers may be written
#  plainly (4000000), with separators (4_000_000), or with a K/M/G suffix (4M).
# ===========================================================================


# ---------------------------------------------------------------------------
#  THE MERSENNE NUMBER(S).  M_p = 2^p - 1, p must be PRIME.
#
#  As of 1.4, the exponent (or exponents) to work on are no longer set here --
#  they live in worktodo_file below (worktodo.txt by default), one per line,
#  processed in order. See that file's own header comment for the format and
#  for the same minimum-exponent note that used to live here.
# ---------------------------------------------------------------------------


# ---------------------------------------------------------------------------
#  WHICH METHOD(S) TO RUN.
#    pm1    (default) P-1 only.
#    pp1    P+1 only. Finds factors P-1 cannot, and vice versa, but at lower
#           yield per unit of GPU time -- a secondary mode, not a replacement.
#    both   P+1 first (cheaper per bound and less likely to succeed), then P-1.
#           NOTE: if P+1 already finds a factor, P-1 still runs to completion
#           afterward -- there is no "stop, a factor is already in hand" check
#           ACROSS methods yet (only within one method's own stage 1 -> stage 2).
#
#  P+1 has its own B1 and B2 model now (`b1`/`b2` below apply to BOTH methods,
#  each optimising its own bounds against the same exponent/bias/tolerance --
#  they are not forced to the same B1 or B2). Still shares P-1's `stage2_d`/
#  `stage2_w` pairing SHAPE, a GPU-memory budget decision, not a cost-model gap.
#
#  P+1 needs a base ("seed"); which seeds find a factor depends on q in a way
#  that cannot be known in advance (roughly half of seeds degenerate into a
#  P-1-shaped computation for a given q), so several are tried per exponent.
# ---------------------------------------------------------------------------
method = both

# A comma-separated LIST of literal seed values (each 3..1000000), NOT a
# count -- "pp1_seeds = 3" tries exactly ONE seed, the value 3, not three
# seeds. For the usual several-seeds coverage write out the list, e.g.
# "pp1_seeds = 3,5,7".
pp1_seeds = 3


# ---------------------------------------------------------------------------
#  STAGE-1 BOUND.
#  P-1 finds a factor q = 2kp+1 whenever k is B1-smooth, i.e. every prime power
#  dividing k is at most B1. Cost is about 1.442 * B1 squarings.
#
#  "auto" picks the B1 that minimises expected total work, using Dickman's rho
#  to weigh the chance of success against the cost. It needs factored_to below.
#
#  With a stage 2 following, "auto" picks a SMALLER B1 than it would for a
#  stage-1-only run -- the second stage catches the near misses, so paying for
#  them twice is waste. Setting `stages = 1` therefore raises the B1 that auto
#  chooses; that is the model working, not a bug.
#
#  Raising b1 later does not throw away the earlier run: see `extend` below.
# ---------------------------------------------------------------------------
b1 = auto


# ---------------------------------------------------------------------------
#  HOW FAR THE EXPONENT HAS ALREADY BEEN TRIAL FACTORED, in bits.
#  Only used to choose "auto" bounds: a factor below this is already known not
#  to exist, so the model does not pay for the chance of finding one.
#  The true value for an exponent is on mersenne.org. "auto" guesses from the
#  size of the exponent, which is a fallback, not a substitute.
# ---------------------------------------------------------------------------
factored_to = 70


# ---------------------------------------------------------------------------
#  HOW MUCH A FACTOR IS WORTH relative to a PRP "composite" result.
#  1.0 = equally valuable. 2.0 = a factor is worth two PRP tests, which pushes
#  "auto" towards larger bounds. Ignored when b1 is set explicitly.
# ---------------------------------------------------------------------------
bias = 2.0


# ---------------------------------------------------------------------------
#  FFT CONFIGURATION.
#
#  "auto" is the recommended setting and is what this ships with.
#
#  How auto works: the engine takes tune.txt's entries in measured-cost order
#  and VERIFIES each one at your exponent before using it, falling through to
#  untuned shapes if every tuned entry fails. The verdict is remembered in
#  fft-verified.txt, so the check costs a few seconds once and nothing on later
#  runs. That verification is not paranoia: a tune on this GPU once produced a
#  tune.txt in which every entry computed WRONG results.
#
#  auto does not stop at the first candidate that verifies: it times two or
#  three and takes the fastest. Shapes that all verify have measured 4.9x apart
#  at one exponent. Untuned candidates are tried smallest-transform-first, and
#  within a size, cheapest arithmetic that still has the precision for this
#  exponent. The wait scales with the transform -- about 7s at the smallest
#  exponent, 15-25s at a wavefront one. Timings land in fft-verified.txt, so a
#  repeat run of the same exponent re-picks the same winner for free -- delete
#  that file to search again.
#
#  Recommended workflow for a new exponent:
#     1. put the exponent in worktodo.txt
#     2. Mp_p-1_gpu.exe --tune noconfig      (optional; targets that exponent
#                                               and prints a verified
#                                               recommendation)
#     3. Mp_p-1_gpu.exe                      (auto picks the best verified one)
#
#  Step 2 is per exponent, not once per GPU. Besides tune.txt it writes
#  Mp_p-1_gpu-tune-config.txt, whose kernel options are specific to the
#  transform they were measured on -- each block records the exponent range it
#  was tuned for, and a run ignores any block that does not cover its own
#  exponent. See MANUAL.md, "Tuning".
#
#  Set an explicit spec (e.g. 1:512:8:256:202) only to override that choice.
#  An explicit spec skips candidate selection but is still verified, unless
#  verify_fft = 0.
# ---------------------------------------------------------------------------
fft = auto


# ---------------------------------------------------------------------------
#  GPU SELECTION.  "auto", or an index from --list-devices.
# ---------------------------------------------------------------------------
device = auto


# ---------------------------------------------------------------------------
#  THE EXPONENT QUEUE AND OUTPUT.
#    worktodo_file      exponents to work on, one per line -- see that file
#    results_file       factors and no-factor results are appended here
#    report_every       squarings between progress-line updates
# ---------------------------------------------------------------------------
worktodo_file = worktodo.txt
results_file = results.txt
report_every = 1000


# ---------------------------------------------------------------------------
#  CPU THREADS FOR PHASE 3 (the gcd).
#
#  VESTIGIAL SINCE 1.8 for the default gcd. The production gcd(x-1, M_p) call
#  now runs through GMP's mpz_gcd, which is single-threaded by design -- this
#  setting has no effect on it, no matter what it's set to. It still applies
#  to this project's OWN half-GCD implementation (gcdHalf), which is no
#  longer used by default but is kept in the tree, fully self-tested, as a
#  reversion path (see Gcd.h). If gcdHalf is ever reinstated as the default,
#  this is what it would consult: "auto" uses every hardware thread; a number
#  leaves cores free. Left at a concrete value rather than "auto" so it's
#  ready to matter again without edits, should that ever happen.
# ---------------------------------------------------------------------------
gcd_threads = 8

checkpoint_seconds = 300


# ---------------------------------------------------------------------------
#  REUSING COMPLETED WORK WHEN YOU RAISE A BOUND.
#
#  On by default. Raising a bound then costs only the difference:
#
#    b1  a completed smaller B1 is extended to the new one, since E(oldB1)
#        divides E(newB1). The saving is exactly oldB1/newB1 of the squarings.
#    b2  a completed stage 2 leaves its accumulator on disk, and the next run
#        walks only (oldB2, newB2]. The accumulator is a product, so the two
#        ranges simply multiply -- and the pairing shape does not have to match.
#
#  The superseded stage-2 file is deleted once the wider one is safely written.
#  Set to 0 to always compute from scratch (useful only for verification).
# ---------------------------------------------------------------------------
extend = 1


# ---------------------------------------------------------------------------
#  AUTOPRIMENET INTEROP.
#
#  worktodo.txt accepts Pfactor=/Pminus1= assignment lines alongside plain
#  exponents. BOTH are P-1 work -- this program does not do trial factoring
#  (bit-depth-range TF, which Prime95 calls "Factor=", a different keyword
#  this program does not recognise) -- they differ only in who picks B1/B2:
#    Pminus1=  PrimeNet already chose B1/B2; use them as-is (see below).
#    Pfactor=  no B1/B2 given -- just how far this exponent has already
#              been trial-factored and what a factor is worth in saved
#              PRP-test credit. The client is expected to pick its own
#              bounds from those two numbers, exactly the role this
#              program's own factored_to/bias already play for "auto".
#  These are two WORK TYPES in the shared worktodo-file format Prime95
#  itself defines, not "AutoPrimeNet writes one, Prime95 writes the other"
#  -- either program can emit either line, chosen by which assignment type
#  PrimeNet handed out, not by which client fetched it. AutoPrimeNet
#  (github.com/tdulcet/AutoPrimeNet) is the maintained tool that speaks
#  this format on your behalf; this program never talks to PrimeNet itself
#  -- AutoPrimeNet is a separate program you run alongside this one,
#  watching the same worktodo.txt/results.txt.
#
#  A Pminus1= line's own B1/B2 -- not this program's cost model -- are used
#  whenever present, on the theory that PrimeNet's assignment is the whole
#  reason to run AutoPrimeNet alongside this program in the first place. A
#  Pfactor= line or a bare exponent has none to honor, and falls back to
#  b1/b2 below (auto by default), same as always.
# ---------------------------------------------------------------------------

# Instead of exiting when worktodo.txt runs dry, wait and keep re-checking it
# -- for an unattended run sitting alongside AutoPrimeNet, so the queue
# running empty for a moment doesn't need a relaunch by hand. Off by default:
# a script or a one-shot manual run expects an empty queue to exit cleanly.
#   wait_for_work        yes | no  (also accepts 1/0, true/false, on/off)
#   wait_poll_seconds     how often to recheck the queue while waiting, in
#                          seconds -- only used when wait_for_work = yes
wait_for_work = no
wait_poll_seconds = 5


# ---------------------------------------------------------------------------
#  HOW HARD TO PUSH THE BOUNDS.
#
#  The expected-cost minimum this program optimises is very FLAT. Measured for
#  M81679223 at TF76: moving from the exact minimum (B1=0.2M, B2=6M) to 4x/8x
#  those bounds costs 2.6% more expected work and nearly DOUBLES the chance of a
#  factor, 2.36% -> 4.39%. Total work is a couple of percent of a PRP test either
#  way. Taking the exact argmin of a curve that flat is false precision: the
#  model's own inputs are not accurate to 2.6%.
#
#  So the choice is made differently: find the cheapest expected cost, then take
#  the HIGHEST-YIELD bounds within this fraction of it.
#
#    0.05  (default) accept up to 5% more expected cost for more factors
#    0     strict argmin -- smallest bounds, lowest yield
#    0.20  push considerably harder
#
#  This is a different knob from `bias`. `bias` says what a factor is worth;
#  this says how much to trust a minimum the model cannot really resolve.
# ---------------------------------------------------------------------------
bounds_tolerance = 0.05


# ---------------------------------------------------------------------------
#  WHICH STAGES TO RUN.
#
#  Stage 1 finds a factor q = 2kp+1 when k is entirely B1-smooth. Stage 2 also
#  finds it when k is B1-smooth APART FROM ONE prime in (B1, B2] -- which is by
#  far the most common near-miss, so stage 2 roughly doubles the yield for
#  comparable work.
#
#    auto   (default) let the bounds model decide. It may still decline stage 2:
#           on a small exponent at bias 1 a PRP test is cheap enough that P-1
#           barely pays for itself, and a second stage does not earn its place.
#    both   always run stage 2, even where the model would not choose it.
#    1      stage 1 only. B1 is then optimised on its own, which makes it LARGER
#           than it would be with a stage 2 following -- with no second stage
#           there is nothing else to catch a near-miss, so it pays to push B1
#           harder.
#
#  `stages` counts stages, so 1 means stage 1 only and 2 means both. If you
#  prefer a yes/no switch use `stage2 = yes | no | auto` instead; the two keys
#  are separate precisely so that neither reads backwards.
# ---------------------------------------------------------------------------
stages = auto

# B2, the stage-2 bound. 'auto' lets the model pick it alongside B1; expect
# something like 30x B1. Ignored (and rejected as a contradiction) if stages = 1.
#
# Raising b2 after a stage 2 has finished walks only the new range, so working an
# exponent upwards in steps costs the same as going straight to the final bound.
# Note that `auto` does NOT know what is already on disk -- if you are stepping
# bounds up deliberately, set b2 explicitly.
b2 = auto

# ---------------------------------------------------------------------------
#  STAGE-2 PAIRING SHAPE.  Leave both on auto unless you are experimenting.
#
#  Stage 2 pairs primes so that one modular multiply covers two of them: with
#  A_m = x^((m*D)^2) and T_j = x^(j^2), the single term A_m - T_j is divisible by
#  both m*D-j and m*D+j. D sets the block size; the window w lets j run up to
#  w*D/2, which gives each prime several candidate slots instead of exactly one
#  and is what lifts the pairing rate from ~1.17 primes per multiply to ~1.6.
#
#  The cost is memory: the T_j table holds phi(D)*w/2 full residues on the GPU.
#  'auto' measures free VRAM and picks the best shape that fits.
# ---------------------------------------------------------------------------
stage2_d = auto
stage2_w = auto


# ---------------------------------------------------------------------------
#  HOLD THE WINDOW OPEN AT EXIT.
#    auto    (default) hold only when launched by double-click, i.e. when the
#            window would otherwise close and destroy the output. Running from
#            an existing shell does not pause, so scripts are unaffected.
#    always  always wait for Enter.
#    never   never wait.
#
#  This applies to every exit path -- results, self-tests, --list-devices and
#  errors alike. One exception: if config.txt itself fails to parse, the setting
#  cannot be read, so "auto" behaviour is used (which still pauses on a
#  double-click, the case that matters).
# ---------------------------------------------------------------------------
pause_on_exit = auto
